/* ---------- 地标：高塔 / 镜湖 / 石桥 ---------- */
const TOWER = { x: -25, z: -24, r: 3.2 };
const POND = { x: 18, z: -26, r: 9 };
const BRIDGE = { x: 18, halfW: 1.9, z0: -38.5, z1: -13.5, h: 1.6 };
const ripples = [];

function heightAt(x, z) {
  if (Math.abs(x - BRIDGE.x) <= BRIDGE.halfW && z > BRIDGE.z0 && z < BRIDGE.z1) {
    const t = (z - BRIDGE.z0) / (BRIDGE.z1 - BRIDGE.z0);
    return Math.sin(Math.PI * t) * BRIDGE.h;
  }
  return 0;
}

function glowSprite() {
  const c = document.createElement('canvas');
  c.width = c.height = 64;
  const g = c.getContext('2d');
  const grd = g.createRadialGradient(32, 32, 0, 32, 32, 32);
  grd.addColorStop(0, 'rgba(255,205,140,.95)');
  grd.addColorStop(.4, 'rgba(255,170,90,.4)');
  grd.addColorStop(1, 'rgba(0,0,0,0)');
  g.fillStyle = grd; g.fillRect(0, 0, 64, 64);
  const t = new THREE.CanvasTexture(c);
  t.colorSpace = THREE.SRGBColorSpace;
  return t;
}

/* 塔 */
(function buildTower() {
  const g = new THREE.Group();
  const stone = new THREE.MeshStandardMaterial({ color: 0x4a4a58, roughness: .88, metalness: .08 });
  const stone2 = new THREE.MeshStandardMaterial({ color: 0x3b3b4a, roughness: .9 });

  const base = new THREE.Mesh(new THREE.CylinderGeometry(3.1, 3.5, 1.2, 16), stone2);
  base.position.y = .6; base.castShadow = true; base.receiveShadow = true;
  g.add(base);

  for (let i = 0; i < 3; i++) {
    const rTop = 2.5 - i * .45, rBot = 2.9 - i * .45, h = 3.2;
    const seg = new THREE.Mesh(new THREE.CylinderGeometry(rTop, rBot, h, 16), i % 2 ? stone2 : stone);
    seg.position.y = 1.2 + h / 2 + i * (h + .15);
    seg.castShadow = true; seg.receiveShadow = true;
    g.add(seg);
    const band = new THREE.Mesh(new THREE.TorusGeometry(rTop + .05, .1, 6, 18),
      new THREE.MeshStandardMaterial({ color: 0xd8b887, roughness: .6, metalness: .3 }));
    band.rotation.x = Math.PI / 2;
    band.position.y = 1.2 + h + i * (h + .15);
    g.add(band);
  }

  const cap = new THREE.Mesh(new THREE.ConeGeometry(2.05, 2.6, 16), stone2);
  cap.position.y = 11.9; cap.castShadow = true;
  g.add(cap);

  const orb = new THREE.Mesh(new THREE.SphereGeometry(.5, 18, 14),
    new THREE.MeshStandardMaterial({ color: 0xfff2c9, emissive: 0xffb45c, emissiveIntensity: 2.4, roughness: .3 }));
  orb.position.y = 13.6;
  g.add(orb);
  const orbLight = new THREE.PointLight(0xffb469, 3.2, 34, 1.8);
  orbLight.position.y = 13.6;
  g.add(orbLight);

  for (let i = 0; i < 6; i++) {
    const a = i / 6 * 6.283 + .3;
    const win = new THREE.Mesh(new THREE.BoxGeometry(.5, .9, .12),
      new THREE.MeshStandardMaterial({ color: 0xffdf9a, emissive: 0xffa95c, emissiveIntensity: 1.6, roughness: .4 }));
    win.position.set(Math.sin(a) * 2.78, 3.4, Math.cos(a) * 2.78);
    win.rotation.y = a;
    g.add(win);
  }

  g.position.set(TOWER.x, 0, TOWER.z);
  g.rotation.y = .6;
  scene.add(g);
  addBlock(TOWER.x, TOWER.z, TOWER.r);
  window.__towerOrb = orb;
})();

/* 镜湖 */
(function buildPond() {
  const sand = new THREE.Mesh(new THREE.RingGeometry(POND.r - .4, POND.r + 2.2, 40),
    new THREE.MeshStandardMaterial({ color: 0x6f6450, roughness: 1 }));
  sand.rotation.x = -Math.PI / 2;
  sand.position.set(POND.x, .02, POND.z);
  sand.receiveShadow = true;
  scene.add(sand);

  const water = new THREE.Mesh(new THREE.CircleGeometry(POND.r, 48),
    new THREE.MeshStandardMaterial({ color: 0x245d7a, roughness: .18, metalness: .55,
      transparent: true, opacity: .72 }));
  water.rotation.x = -Math.PI / 2;
  water.position.set(POND.x, .06, POND.z);
  scene.add(water);

  const glow = new THREE.PointLight(0x63b8ff, 1.1, 26, 1.8);
  glow.position.set(POND.x, 1.2, POND.z);
  scene.add(glow);

  for (let i = 0; i < 4; i++) {
    const m = new THREE.Mesh(new THREE.RingGeometry(.7, .95, 30),
      new THREE.MeshBasicMaterial({ color: 0x9fe8ff, transparent: true, opacity: .3,
        side: THREE.DoubleSide, depthWrite: false }));
    m.rotation.x = -Math.PI / 2;
    m.position.set(POND.x + (Math.random() - .5) * 8, .08, POND.z + (Math.random() - .5) * 8);
    scene.add(m);
    ripples.push({ mesh: m, t: Math.random() });
  }

  for (let i = 0; i < 16; i++) {
    const a = Math.random() * 6.283, rr = POND.r - .2 + Math.random() * 1.2;
    const reed = new THREE.Mesh(new THREE.CylinderGeometry(.03, .05, .9 + Math.random() * .7, 5),
      new THREE.MeshStandardMaterial({ color: 0x3f6b4a, roughness: .9 }));
    reed.position.set(POND.x + Math.sin(a) * rr, .5, POND.z + Math.cos(a) * rr);
    reed.rotation.z = (Math.random() - .5) * .3;
    scene.add(reed);
  }
})();

/* 石桥 */
(function buildBridge() {
  const g = new THREE.Group();
  const slabMat = new THREE.MeshStandardMaterial({ color: 0x8b8b96, roughness: .92, flatShading: true });
  const railMat = new THREE.MeshStandardMaterial({ color: 0x6d6d7a, roughness: .9 });
  const len = BRIDGE.z1 - BRIDGE.z0;
  const steps = 22;
  for (let i = 0; i < steps; i++) {
    const t = (i + .5) / steps;
    const z = BRIDGE.z0 + t * len;
    const y = Math.sin(Math.PI * t) * BRIDGE.h;
    const slab = new THREE.Mesh(new THREE.BoxGeometry(BRIDGE.halfW * 2 - .1, .18, len / steps + .06), slabMat);
    slab.position.set(BRIDGE.x, y - .06, z);
    slab.castShadow = true; slab.receiveShadow = true;
    g.add(slab);

    if (i % 2 === 0) {
      for (const side of [-1, 1]) {
        const rail = new THREE.Mesh(new THREE.BoxGeometry(.16, 1, len / steps * 1.6), railMat);
        rail.position.set(BRIDGE.x + side * (BRIDGE.halfW - .12), y + .45, z);
        rail.castShadow = true;
        g.add(rail);
      }
    }
  }
  scene.add(g);
})();

const towerHalo = new THREE.Sprite(new THREE.SpriteMaterial({
  map: glowSprite(), transparent: true, opacity: .35, blending: THREE.AdditiveBlending, depthWrite: false
}));
towerHalo.scale.set(9, 9, 1);
towerHalo.position.set(TOWER.x, 13.6, TOWER.z);
scene.add(towerHalo);
